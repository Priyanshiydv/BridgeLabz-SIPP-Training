    String movieName;
    String seatNumber;
    double price;
    boolean isBooked;
}

    method cancelTicket(ticket) {
        mark ticket.seatNumber as available
        ticket.isBooked = false
        print "Ticket cancelled."
    }
}

// Customer class
class Customer {
    String name

    method bookTicket(theatre, movie, seat) {
        Ticket t = theatre.issueTicket(movie, seat)
        if (t != null) print "Booking successful."
    }

    method cancelTicket(theatre, ticket) {
        theatre.cancelTicket(ticket)
    }
}
