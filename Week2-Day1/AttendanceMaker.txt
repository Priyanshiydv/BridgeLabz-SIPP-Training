class AttendanceSystem {
    char[][] attendance   // rows=students, cols=days

    method markAttendance(studentId, day, mark) {
        attendance[studentId][day] = mark  // 'P' or 'A'
    }

    method updateAttendance(studentId, day, mark) {
        attendance[studentId][day] = mark
    }

    method generateReport(studentId) {
        count 'P' in row of studentId
        print "Student present X days"
    }
}

// Example run
main() {
    AttendanceSystem sys
    sys.markAttendance(1, 1, 'P')
    sys.markAttendance(1, 2, 'P')
    sys.markAttendance(1, 3, 'A')
    sys.updateAttendance(1, 3, 'P')
    sys.generateReport(1)
}
